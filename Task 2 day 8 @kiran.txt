Task 1: Create database
CREATE DATABASE ecommerce_db;
Task 2: Use Database
USE ecommerce_db;
Task 3: Create products table
CREATE TABLE products (
    product_id INT PRIMARY KEY,
    product_name VARCHAR(100),
    category VARCHAR(50),
    brand VARCHAR(50),
    price DECIMAL(10,2),
    quantity INT,
    city VARCHAR(50),
    status VARCHAR(20)
);
Task 4: display table structure
DESC products;
